cmake_minimum_required(VERSION 3.16)
project(MiniHV C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
file(GLOB_RECURSE MINIHV_SOURCES CONFIGURE_DEPENDS "src/*.c")
add_executable(minihv ${MINIHV_SOURCES})
target_include_directories(minihv PRIVATE src)
if(MSVC)
  target_compile_options(minihv PRIVATE /W4 /WX)
else()
  target_compile_options(minihv PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
